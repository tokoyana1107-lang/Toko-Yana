Panduan Real-Time Google Sheets

Hubungkan TOKO YANA ke Google Sheets milik Anda

  1. Buka Google Sheets dan buat spreadsheet baru (nama file bebas, misalnya Database TOKO YANA).
  2. Klik menu Ekstensi > Apps Script, hapus kode bawaan, lalu salin kode di bawah ini dan klik Save. Skrip akan otomatis membuatkan sheet terpisah Data Orang, Transaksi, dan Pengaturan.
  3. Klik Deploy > New deployment, pilih tipe Web app, atur Execute as ke Me dan Who has access ke Anyone, lalu klik Deploy.
  4. Salin Web app URL yang berakhiran /exec, tempelkan ke kolom URL di menu Pengaturan & Sandi, lalu klik Simpan & Tes Koneksi.
const SHEET_CUSTOMERS = 'Data Orang';
const SHEET_TRANSACTIONS = 'Transaksi';
const SHEET_SETTINGS = 'Pengaturan';

function getOrCreateSheet_(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  const firstRow = sheet.getRange(1, 1, 1, headers.length).getValues()[0];
  if (!headers.every((h, idx) => String(firstRow[idx] || '').trim() === h)) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  }
  return sheet;
}

function readCustomers_() {
  const headers = ['ID', 'Nama Lengkap', 'No. HP / WA', 'NIK', 'Alamat', 'Catatan', 'Diperbarui'];
  const sheet = getOrCreateSheet_(SHEET_CUSTOMERS, headers);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  return sheet.getRange(2, 1, lastRow - 1, headers.length).getValues()
    .filter(r => String(r[0] || '').trim() || String(r[1] || '').trim())
    .map(r => ({
      id: String(r[0] || '').trim(),
      name: String(r[1] || '').trim(),
      phone: String(r[2] || '').trim(),
      nik: String(r[3] || '').trim(),
      address: String(r[4] || '').trim(),
      notes: String(r[5] || '').trim(),
      photoBase64: ''
    }));
}

function writeCustomers_(customers) {
  const headers = ['ID', 'Nama Lengkap', 'No. HP / WA', 'NIK', 'Alamat', 'Catatan', 'Diperbarui'];
  const sheet = getOrCreateSheet_(SHEET_CUSTOMERS, headers);
  const now = new Date().toISOString();
  const rows = (Array.isArray(customers) ? customers : []).map(c => [
    String(c.id || ''), String(c.name || ''), String(c.phone || ''),
    String(c.nik || ''), String(c.address || ''), String(c.notes || ''), now
  ]);
  const maxRows = Math.max(sheet.getMaxRows() - 1, rows.length, 1);
  sheet.getRange(2, 1, maxRows, headers.length).clearContent();
  if (rows.length > 0) sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function readTransactions_() {
  const headers = [
    'ID Nota', 'Jenis', 'ID Orang', 'Nama Pelanggan', 'No. HP',
    'Barang / Rincian', 'Total Harga', 'DP / Awal', 'Tenor',
    'Tanggal', 'Jatuh Tempo', 'Catatan', 'Riwayat Bayar JSON', 'Diperbarui'
  ];
  const sheet = getOrCreateSheet_(SHEET_TRANSACTIONS, headers);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  return sheet.getRange(2, 1, lastRow - 1, headers.length).getValues()
    .filter(r => String(r[0] || '').trim() || String(r[5] || '').trim())
    .map(r => {
      let payments = [];
      try { payments = JSON.parse(String(r[12] || '[]')) || []; } catch (e) {}
      const personId = String(r[2] || '').trim() || null;
      return {
        id: String(r[0] || '').trim(),
        type: String(r[1] || 'UTANG').trim(),
        personId: personId,
        customerId: personId,
        customerName: String(r[3] || '').trim(),
        customerPhone: String(r[4] || '').trim(),
        itemName: String(r[5] || '').trim(),
        totalAmount: Number(r[6]) || 0,
        dpAmount: Number(r[7]) || 0,
        tenor: Number(r[8]) || 1,
        date: String(r[9] || '').trim(),
        dueDate: String(r[10] || '').trim(),
        notes: String(r[11] || '').trim(),
        payments: Array.isArray(payments) ? payments : []
      };
    });
}

function writeTransactions_(transactions) {
  const headers = [
    'ID Nota', 'Jenis', 'ID Orang', 'Nama Pelanggan', 'No. HP',
    'Barang / Rincian', 'Total Harga', 'DP / Awal', 'Tenor',
    'Tanggal', 'Jatuh Tempo', 'Catatan', 'Riwayat Bayar JSON', 'Diperbarui'
  ];
  const sheet = getOrCreateSheet_(SHEET_TRANSACTIONS, headers);
  const now = new Date().toISOString();
  const rows = (Array.isArray(transactions) ? transactions : []).map(t => [
    String(t.id || ''), String(t.type || 'UTANG'), String(t.personId || t.customerId || ''),
    String(t.customerName || ''), String(t.customerPhone || ''), String(t.itemName || ''),
    Number(t.totalAmount) || 0, Number(t.dpAmount) || 0, Number(t.tenor) || 1,
    String(t.date || ''), String(t.dueDate || ''), String(t.notes || ''),
    JSON.stringify(Array.isArray(t.payments) ? t.payments : []), now
  ]);
  const maxRows = Math.max(sheet.getMaxRows() - 1, rows.length, 1);
  sheet.getRange(2, 1, maxRows, headers.length).clearContent();
  if (rows.length > 0) sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function readSettings_() {
  const headers = ['Pengaturan', 'Nilai', 'Diperbarui'];
  const sheet = getOrCreateSheet_(SHEET_SETTINGS, headers);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  const rows = sheet.getRange(2, 1, lastRow - 1, headers.length).getValues();
  const obj = {};
  let hasAny = false;
  rows.forEach(r => {
    const k = String(r[0] || '').trim();
    if (!k) return;
    hasAny = true;
    obj[k] = k === 'lockEnabled' ? (r[1] === true || String(r[1]).toLowerCase() === 'true') : r[1];
  });
  return hasAny ? obj : null;
}

function writeSettings_(settings) {
  const headers = ['Pengaturan', 'Nilai', 'Diperbarui'];
  const sheet = getOrCreateSheet_(SHEET_SETTINGS, headers);
  const now = new Date().toISOString();
  const safe = Object.assign({}, settings || {});
  const keys = ['storeName', 'slogan', 'phone', 'address', 'password', 'lockEnabled', 'waTemplate'];
  const rows = keys.map(k => [k, safe[k] !== undefined ? safe[k] : '', now]);
  const maxRows = Math.max(sheet.getMaxRows() - 1, rows.length, 1);
  sheet.getRange(2, 1, maxRows, headers.length).clearContent();
  if (rows.length > 0) sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function doGet() {
  const data = {
    customers: readCustomers_(),
    transactions: readTransactions_(),
    settings: readSettings_() || {}
  };
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (Array.isArray(body.customers)) writeCustomers_(body.customers);
    if (Array.isArray(body.transactions)) writeTransactions_(body.transactions);
    if (body.settings && typeof body.settings === 'object') writeSettings_(body.settings);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, updatedAt: new Date().toISOString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
